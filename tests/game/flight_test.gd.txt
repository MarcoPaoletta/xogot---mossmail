extends RefCounted
## Flight test: an updraft mushroom lifts the glider, the hills raise the
## cruise line, and gliding through the sky ring completes Sky mail.

func _run() -> void:
	var tree: SceneTree = Engine.get_main_loop() as SceneTree
	var main: Node = tree.current_scene
	var glider: Glider = main.get_node("Glider") as Glider
	var director: DeliveryDirector = main.get_node("DeliveryDirector") as DeliveryDirector
	var terrain: TerrainField = main.get_node("Terrain") as TerrainField
	main.call("_start_shift")
	await _frames(tree, 2)
	director.set("_pattern_index", 3)
	director.set("_delay_left", 0.0)
	await _frames(tree, 3)
	_check("sky mail task started", director.get_task_title() == "Sky mail")
	var pad: Node3D = director.get("_pad") as Node3D
	_check("updraft mushroom targeted", pad != null)
	if pad == null:
		return
	var cruise_y: float = terrain.height_at(pad.global_position.x, pad.global_position.z) + glider.cruise_height
	glider.global_position = Vector3(pad.global_position.x, cruise_y, pad.global_position.z)
	await _frames(tree, 4)
	_check("updraft caught", glider.is_in_updraft())
	var start_y: float = glider.global_position.y
	await _frames(tree, 90)
	var climbed: float = glider.global_position.y - start_y
	print("climbed ", snappedf(climbed, 0.1), " m in 1.5 s")
	_check("updraft lifts the glider", climbed > 5.0)
	var ring_pos: Vector3 = director.get_target_position()
	print("ring above ground ", snappedf(ring_pos.y - terrain.height_at(ring_pos.x, ring_pos.z), 0.1))
	glider.global_position = ring_pos
	await _frames(tree, 4)
	_check("sky ring collected", director.deliveries == 1)
	print("score ", director.score)
	var samples: Array = []
	for x: int in [-400, -150, 0, 230, 560]:
		samples.append(snappedf(terrain.height_at(float(x), float(x) * 0.7), 0.1))
	print("terrain samples ", samples)
	director.stop_shift()


func _frames(tree: SceneTree, count: int) -> void:
	for i: int in count:
		await tree.physics_frame


func _check(label: String, ok: bool) -> void:
	print(("PASS " if ok else "FAIL ") + label)
